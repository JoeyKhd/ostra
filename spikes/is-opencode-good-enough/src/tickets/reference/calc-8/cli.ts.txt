import { createCalculator } from "./calculator.ts";
import { CalcError } from "./errors.ts";

export type CliIo = {
  readonly out: (line: string) => void;
  readonly err: (line: string) => void;
};

const USAGE = `Usage: calc [--thousands] <expression>...

Evaluates each expression and prints the result on its own line.

Examples:
  calc "1 + 2 * 3"
  calc --thousands "1000 * 1000"`;

/** Runs the command line and returns its exit code: 0 on success, 1 for a wrong expression, 2 for wrong use. */
export const runCli = (args: readonly string[], io: CliIo): number => {
  const expressions = args.filter((arg) => arg !== "--thousands");
  const unknownOption = expressions.find((arg) => arg.startsWith("--") && arg !== "--help");

  if (args.includes("--help") || args.includes("-h")) {
    io.out(USAGE);
    return 0;
  }
  if (unknownOption !== undefined) {
    io.err(`error: unknown option ${unknownOption}`);
    return 2;
  }
  if (expressions.length === 0) {
    io.err(USAGE);
    return 2;
  }
  const calculator = createCalculator({ format: { thousands: args.includes("--thousands") } });
  for (const expression of expressions) {
    try {
      io.out(calculator.run(expression).text);
    } catch (error) {
      if (!(error instanceof CalcError)) throw error;
      io.err(`error: ${error.message}`);
      return 1;
    }
  }
  return 0;
};

if (import.meta.main) {
  process.exitCode = runCli(process.argv.slice(2), { out: console.log, err: console.error });
}
